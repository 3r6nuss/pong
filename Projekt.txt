async function loadHighscoresFromDatabase() {
      const response = await fetch("/api/highscores");
      const highscores = await response.json();

      players.left.wins = 0;
      players.right.wins = 0;

      const leftEntry = highscores.find(
        (entry) => entry.playerName === players.left.name
      );

      const rightEntry = highscores.find(
        (entry) => entry.playerName === players.right.name
      );

      if (leftEntry) {
        players.left.wins = leftEntry.wins;
      }

      if (rightEntry) {
        players.right.wins = rightEntry.wins;
      }
    }

 async function saveWinToDatabase(playerName) {
      const response = await fetch("/api/highscores/win", {
        method: "POST",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ playerName })
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.error || "Sieg konnte nicht gespeichert werden.");
      }

      return data;
    }

async function handleRoundWin(side) {
      state.roundOver = true;
      state.winnerText = players[side].name + " gewinnt die Runde!";

      resetKeys();

      try {
        const result = await saveWinToDatabase(players[side].name);

        const leftEntry = result.highscores.find(
          (entry) => entry.playerName === players.left.name
        );

        const rightEntry = result.highscores.find(
          (entry) => entry.playerName === players.right.name
        );

        players.left.wins = leftEntry ? leftEntry.wins : 0;
        players.right.wins = rightEntry ? rightEntry.wins : 0;
      } catch (error) {
        alert("Fehler beim Speichern in der Datenbank: " + error.message);
      }

      alert(
        players[side].name + " gewinnt die Runde!\n\n" +
        "Aktueller Highscore:\n" +
        players.left.name + ": " + players.left.wins + " Sieg(e)\n" +
        players.right.name + ": " + players.right.wins + " Sieg(e)\n\n" +
        "Drücke danach LEERTASTE für die nächste Runde."
      );
    }
